62inline scope_t scope(std::string place, unsigned int line) {
63#ifndef LOGCPP_LEAVE_SCOPE_DIRS_PREFIX
64 unsigned int dir_prefix_end = 0;
65#ifdef __unix__
66 dir_prefix_end = place.find_last_of('/');
67#elif _WIN32
68 dir_prefix_end = place.find_last_of('\\');
69#endif
70 if ( dir_prefix_end != std::string::npos && dir_prefix_end != 0 ) {
71 place = place.substr(dir_prefix_end+1);
72 }
73#endif
74 return scope_t( place, line);
75}

86inline const std::string timestr() {
87 std::time_t rawtime = std::time(0);
88 struct std::tm* tinfo;
89 tinfo = std::localtime(&rawtime);
90 char buf[22];
91 std::strftime(buf, 22, "%F_%T", tinfo);
